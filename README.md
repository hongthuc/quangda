            const driveId = imgEl.getAttribute('data-drive-id');
            const fallbackUrl = imgEl.getAttribute('data-fallback');

                imgEl.setAttribute('data-tried-lh3', 'true');
                imgEl.src = 'https://lh3.googleusercontent.com/d/' + driveId + '=w1000';
            } else if (fallbackUrl && imgEl.src !== fallbackUrl) {
                imgEl.src = fallbackUrl;
            }
        }

        /* Tự động trượt 2 Banner kích thước bằng nhau */
        let activeSlide = 0;
        const totalSlides = 2;
        let bannerInterval = null;

        function updateBannerUI() {
            for (let i = 0; i < totalSlides; i++) {
                const slideEl = document.getElementById('banner-slide-' + i);
                const dotEl = document.getElementById('banner-dot-' + i);
                if (i === activeSlide) {
                    slideEl.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
                    slideEl.classList.add('opacity-100', 'z-10');
                    dotEl.className = 'w-6 h-2 rounded-full bg-qdsand-400 transition-all';
                } else {
                    slideEl.classList.remove('opacity-100', 'z-10');
                    slideEl.classList.add('opacity-0', 'z-0', 'pointer-events-none');
                    dotEl.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
                }
            }
        }

        function changeBannerSlide(step) {
            activeSlide = (activeSlide + step + totalSlides) % totalSlides;
            updateBannerUI();
            resetBannerTimer();
        }

        function goToBannerSlide(index) {
            activeSlide = index;
            updateBannerUI();
            resetBannerTimer();
        }

        function resetBannerTimer() {
            if (bannerInterval) clearInterval(bannerInterval);
            bannerInterval = setInterval(() => {
                activeSlide = (activeSlide + 1) % totalSlides;
                updateBannerUI();
            }, 5000);
        }

        window.onload = function() {
            renderProducts();
            resetBannerTimer();
        };
